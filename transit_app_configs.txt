# 10. Nginx Reverse Proxy (Production)
# -------------------------------------
# Dosya: /etc/nginx/sites-available/gtfs-rt

server {
    listen 80;
    server_name gtfs.samulas.com.tr;
    
    # HTTPS redirect (production için)
    # return 301 https://$server_name$request_uri;
    
    location /gtfs-rt/ {
        proxy_pass http://localhost:8000/gtfs-rt/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        
        # Caching
        proxy_cache gtfs_cache;
        proxy_cache_valid 200 15s;
        proxy_cache_key $uri;
    }
    
    location /gtfs-static {
        proxy_pass http://localhost:8000/gtfs-static;
        
        # Long cache for static
        proxy_cache gtfs_cache;
        proxy_cache_valid 200 24h;
    }
    
    # Health check
    location /health {
        proxy_pass http://localhost:8000/gtfs-rt/stats;
        access_log off;
    }
}

# Cache configuration
proxy_cache_path /var/cache/nginx/gtfs levels=1:2 keys_zone=gtfs_cache:10m max_size=100m;

# Dosya: docker-compose.yml

version: '3.8'

services:
  samsun-transit:
    build: .
    ports:
      - "8000:8000"
    volumes:
      - ./data:/app/data
      - ./logs:/app/logs
    environment:
      - TZ=Europe/Istanbul
      - LOG_LEVEL=INFO
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/gtfs-rt/stats"]
      interval: 30s
      timeout: 10s
      retries: 3
  
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
      - ./ssl:/etc/nginx/ssl:ro
    depends_on:
      - samsun-transit
    restart: unless-stopped


# 12. Systemd Service (Linux Server)
# -----------------------------------
# Dosya: /etc/systemd/system/samsun-transit.service

[Unit]
Description=Samsun Transit GTFS-RT Service
After=network.target

[Service]
Type=simple
User=transit
Group=transit
WorkingDirectory=/opt/samsun-transit
ExecStart=/usr/bin/python3 /opt/samsun-transit/samsun.py
Restart=always
RestartSec=10
StandardOutput=append:/var/log/samsun-transit/access.log
StandardError=append:/var/log/samsun-transit/error.log

[Install]
WantedBy=multi-user.target

    maxconn 4096

    mode http
    timeout connect 5000ms
    timeout client 50000ms
    timeout server 50000ms

frontend gtfs_frontend
    bind *:8000
    default_backend gtfs_backend

backend gtfs_backend
    balance roundrobin
    option httpchk GET /gtfs-rt/stats
    
    server transit1 localhost:8001 check
    server transit2 localhost:8002 check
    server transit3 localhost:8003 check
